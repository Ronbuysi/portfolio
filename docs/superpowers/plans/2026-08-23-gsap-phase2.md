const state = { x: 0, velocity: 0, lastY: window.scrollY, dir: -1, hover: false }
const skewTo = gsap.quickTo(track, 'skewX', { duration: 0.45, ease: 'power2.out' })

const tick = (time, deltaTime) => {
  const delta = Math.min(deltaTime / 16.7, 4)
  const y = window.scrollY
  const scrolled = y - state.lastY
  state.lastY = y
  state.velocity += (scrolled * 0.05 - state.velocity) * 0.08
  if (state.velocity > 0.12) state.dir = -1
  else if (state.velocity < -0.12) state.dir = 1
  skewTo(gsap.utils.clamp(-7, 7, state.velocity * 2.4))
  state.x -= (0.55 + Math.abs(state.velocity)) * delta * state.dir * (state.hover ? 0 : 1)
  const half = track.scrollWidth / 2
  if (half > 0) {
    if (state.x <= -half) state.x += half
    if (state.x > 0) state.x -= half
  }
  track.style.transform = `translate3d(${state.x.toFixed(2)}px, 0, 0)`
}

const enter = () => { state.hover = true }
const leave = () => { state.hover = false }
track.parentElement.addEventListener('pointerenter', enter)
track.parentElement.addEventListener('pointerleave', leave)
gsap.ticker.add(tick)
return () => {
  gsap.ticker.remove(tick)
  track.parentElement.removeEventListener('pointerenter', enter)
  track.parentElement.removeEventListener('pointerleave', leave)
  gsap.set(track, { clearProps: 'all' })
}

```jsx
import { useEffect, useRef, useState } from 'react'
import { gsap } from '../../motion/gsapSetup'

export default function CountUp({ value = '', duration = 900 }) {
  const ref = useRef(null)
  const [initial] = useState(() => value.replace(/\d/g, '0'))

  useEffect(() => {
    const el = ref.current
    if (!el) return undefined
    const end = parseInt(value, 10)
    if (Number.isNaN(end)) { el.textContent = value; return undefined }
    const width = value.length
    const forced = document.documentElement.classList.contains('motion-forced')
    if (!forced && window.matchMedia?.('(prefers-reduced-motion: reduce)')?.matches) { el.textContent = value; return undefined }
    if (typeof IntersectionObserver === 'undefined') { el.textContent = value; return undefined }

    const state = { n: 0 }
    const render = () => { el.textContent = String(Math.round(state.n)).padStart(width, '0') }
    const tween = gsap.to(state, {
      n: end,
      duration: duration / 1000,
      ease: 'back.out(1.25)',
      paused: true,
      onUpdate: render,
    })
    const observer = new IntersectionObserver(([entry]) => {
      if (!entry.isIntersecting) return
      tween.play()
      observer.disconnect()
    }, { threshold: 0.4 })
    observer.observe(el)

    return () => { tween.kill(); observer.disconnect() }
  }, [value, duration])

  return <span ref={ref}>{initial}</span>
}
```

```js
import { gsap, useGSAP, motionAllowed } from '../motion/gsapSetup'

export default function useMagneticLinks(rootRef, selector = 'a[href], button') {
  useGSAP((context, contextSafe) => {
    const root = rootRef.current
    if (!root || !motionAllowed()) return undefined
    if (!window.matchMedia?.('(pointer: fine)')?.matches) return undefined

    const cleanups = Array.from(root.querySelectorAll(selector)).map((link) => {
      const qx = gsap.quickTo(link, 'x', { duration: 0.4, ease: 'power3.out' })
      const qy = gsap.quickTo(link, 'y', { duration: 0.4, ease: 'power3.out' })
      const onMove = contextSafe((event) => {
        const rect = link.getBoundingClientRect()
        qx(gsap.utils.clamp(-10, 10, (event.clientX - rect.left - rect.width / 2) * 0.25))
        qy(gsap.utils.clamp(-8, 8, (event.clientY - rect.top - rect.height / 2) * 0.35))
      })
      const onLeave = contextSafe(() => { qx(0); qy(0) })
      link.addEventListener('pointermove', onMove)
      link.addEventListener('pointerleave', onLeave)
      return () => {
        link.removeEventListener('pointermove', onMove)
        link.removeEventListener('pointerleave', onLeave)
      }
    })

    return () => cleanups.forEach((fn) => fn())
  }, { scope: rootRef })
}
```

import { gsap, useGSAP, motionAllowed } from '../motion/gsapSetup'

export default function useTiltCards(rootRef, selector, maxTilt = 4) {
  useGSAP((context, contextSafe) => {
    const root = rootRef.current
    if (!root || !motionAllowed()) return undefined
    if (!window.matchMedia?.('(pointer: fine)')?.matches) return undefined

    const cleanups = Array.from(root.querySelectorAll(selector)).map((card) => {
      gsap.set(card, { transformPerspective: 700 })
      const rx = gsap.quickTo(card, 'rotationX', { duration: 0.45, ease: 'power2.out' })
      const ry = gsap.quickTo(card, 'rotationY', { duration: 0.45, ease: 'power2.out' })
      const onMove = contextSafe((event) => {
        const rect = card.getBoundingClientRect()
        const px = (event.clientX - rect.left) / rect.width - 0.5
        const py = (event.clientY - rect.top) / rect.height - 0.5
        ry(px * maxTilt * 2)
        rx(-py * maxTilt * 2)
      })
      const onLeave = contextSafe(() => { rx(0); ry(0) })
      card.addEventListener('pointermove', onMove)
      card.addEventListener('pointerleave', onLeave)
      return () => {
        card.removeEventListener('pointermove', onMove)
        card.removeEventListener('pointerleave', onLeave)
      }
    })

    return () => cleanups.forEach((fn) => fn())
  }, { scope: rootRef })
}

- [ ] **Step 1: 重写 Preloader.jsx**

```jsx
import { useEffect, useRef, useState } from 'react'
import { gsap, useGSAP, motionAllowed } from '../motion/gsapSetup'

export default function Preloader({ done }) {
  const [gone, setGone] = useState(false)
  const panelRef = useRef(null)
  const veilRef = useRef(null)
  const numRef = useRef(null)
  const barRef = useRef(null)

  // count 0 -> 100 while visible
  useGSAP(() => {
    if (done || gone) return undefined
    const state = { n: 0 }
    const render = () => {
      const v = Math.round(state.n)
      if (numRef.current) numRef.current.textContent = `${v}%`
      if (barRef.current) barRef.current.style.width = `${v}%`
    }
    render()
    const tween = gsap.to(state, { n: 100, duration: 1.05, ease: 'power3.out', onUpdate: render })
    return () => { tween.kill() }
  }, { dependencies: [done], revertOnUpdate: true })

  // exit choreography: acid veil leads, dark panel follows
  useGSAP(() => {
    if (!done || gone) return undefined
    if (!motionAllowed()) return undefined
    const root = panelRef.current?.parentElement
    root?.classList.add('gsap-exit')
    const tl = gsap.timeline({ onComplete: () => setGone(true) })
      .to(veilRef.current, { yPercent: -101, duration: 0.72, ease: 'expo.inOut' }, 0)
      .to(panelRef.current, { yPercent: -101, duration: 0.78, ease: 'power4.inOut' }, 0.1)
    return () => { tl.kill() }
  }, { dependencies: [done], revertOnUpdate: true })

  // reduced-motion fallback keeps the CSS class-based lift and a timed unmount
  useEffect(() => {
    if (!done || gone) return undefined
    if (motionAllowed()) return undefined
    const timer = setTimeout(() => setGone(true), 720)
    return () => clearTimeout(timer)
  }, [done, gone])

  if (gone) return null

  return <div className={'preloader' + (done ? ' is-done' : '')} aria-hidden="true">
    <div className="preloader__panel" ref={panelRef}>
      <span className="preloader__word">WANG CHENGCHENG</span>
      <span className="preloader__num" ref={numRef}>0%</span>
      <span className="preloader__bar"><i ref={barRef} style={{ width: '0%' }} /></span>
    </div>
    <span className="preloader__veil" ref={veilRef} />
  </div>
}
```

let pendingState = null

export function captureFlipSource(el) {
  pendingState = el ? Flip.getState(el) : null
}

export function peekPendingFlip() {
  return pendingState !== null
}

export function consumeOpenFlip(target, vars = {}) {
  if (!pendingState) return false
  const state = pendingState
  pendingState = null
  Flip.from(state, { target, scale: true, duration: 0.65, ease: 'power3.inOut', ...vars })
  return true
}

export function clearFlipSource() {
  pendingState = null
}
